async function obtenerTasaBCV() { estado.textContent = "Consultando portal oficial BCV..."; // Proxies públicos para leer el HTML oficial de bcv.org.ve sin bloqueo CORS const proxies = [ "https://api.allorigins.win/raw?url=" + encodeURIComponent("https://www.bcv.org.ve/"), "https://corsproxy.io/?" + encodeURIComponent("https://www.bcv.org.ve/") ]; for (const url of proxies) { try { const resp = await fetch(url, { cache: "no-store" }); if (!resp.ok) continue; const html = await resp.text(); const doc = new DOMParser().parseFromString(html, "text/html"); // En bcv.org.ve, el valor del USD está dentro del bloque con id="dolar" const textoDolar = doc.querySelector("#dolar strong")?.textContent?.trim(); if (!textoDolar) continue; // Convierte "924,35340000" a número y lo formatea a 4 decimales const tasaNum = parsearNumero(textoDolar); if (tasaNum > 0) { inputTasa.value = tasaNum.toFixed(4).replace(".", ","); actualizar(); estado.textContent = "Tasa sincronizada directo desde BCV."; return; } } catch (e) { // Pasa al siguiente proxy si este falla } } // Si ambos proxies fallan o bcv.org.ve no responde, mantiene el respaldo if (!inputTasa.value) { inputTasa.value = TASA_RESPALDO_BCV; } actualizar(); estado.textContent = "No se pudo conectar al BCV; usando tasa de respaldo."; }
